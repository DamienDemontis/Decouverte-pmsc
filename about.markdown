function updateCampusInfo() {
  const select = document.getElementById('campus-select');
  const selectedCampus = select.value;
  
  if (!selectedCampus || !campusData[selectedCampus]) {
    return;
  }
  
  const campus = campusData[selectedCampus];
  
  // Mettre à jour l'email
  const emailLink = document.getElementById('campus-email');
  emailLink.textContent = campus.email;
  emailLink.href = `mailto:${campus.email}`;
  
  // Mettre à jour le téléphone
  const phoneLink = document.getElementById('campus-phone');
  if (campus.phone === "N/A") {
    phoneLink.textContent = "Non disponible";
    phoneLink.href = "#";
    phoneLink.style.pointerEvents = "none";
    phoneLink.style.color = "#9ca3af";
  } else {
    phoneLink.textContent = campus.phone;
    phoneLink.href = `tel:${campus.phoneLink}`;
    phoneLink.style.pointerEvents = "auto";
    phoneLink.style.color = "";
  }
  
  // Mettre à jour l'adresse
  const addressSpan = document.getElementById('campus-address');
  addressSpan.textContent = campus.address;
}

// Initialiser avec Paris sélectionné
document.addEventListener('DOMContentLoaded', function() {
  updateCampusInfo();
});
